package org.apache.kafka.server.config;

/**
 * Represents all the entities that can be configured.
 */
public enum ConfigType {
    TOPIC("topics"),
    CLIENT("clients"),
    USER("users"),
    BROKER("brokers"),
    IP("ips"),
    CLIENT_METRICS("client-metrics"),
    GROUP("groups");

    private final String value;

    ConfigType(String value) {
        this.value = value;
    }

    public String value() {
        return value;
    }
}
